    \def\Rbg{0.65}
    \def\xstepbg{1.5*\Rbg}
    \def\ystepbg{sqrt(3)*\Rbg}

    \foreach \i in {-12,...,12} {
        \foreach \j in {-4,...,4} {

        \pgfmathsetmacro{\x}{\i * \xstepbg}
        \pgfmathsetmacro{\y}{\j * \ystepbg + mod(\i,2)*\ystepbg/2}

        \begin{scope}[shift={(\x,\y)}]
            \filldraw[fill=gray!5, draw=gray!60, line width=0.3pt, fill opacity=0.05]
            (0:\Rbg) \foreach \a in {60,120,...,300} { -- (\a:\Rbg) } -- cycle;

        }

    \def\ystep{sqrt(3)*\R}

    % Mapeamento de índices para símbolos
    \newcommand{\getsymbol}[1]{%
    \ifcase#1 \alpha%
    \or \beta%
    \or \gamma%
    \or \delta%
    \or \epsilon%
    \or \lambda%
    \or \mu%
    \or \pi%
    \or \sigma%
    \or \theta%
    \or \omega%
    \or \land%
    \or \lor%
    \or \neg%
    \or \Rightarrow%
    \or \Leftrightarrow%
    \or e%
    \or \phi%
    \or \infty%
    \else \relax \fi%
    }   

    \foreach \i in {-10,...,10} {
        \foreach \j in {-2,...,2} {

      \pgfmathsetmacro{\x}{\i * \xstep}
      \pgfmathsetmacro{\y}{\j * \ystep + mod(\i,2)*\ystep/2}

      % Define se o hexágono é dourado (~35%)
      \pgfmathtruncatemacro{\isdourado}{(mod(\i+\j,3)==0 || mod(\i*2+\j,7)==0) ? 1 : 0}

      \ifnum\isdourado=1
        % \def\fillcolor{customgold}
        \def\fillcolor{accentColor}
        \def\drawcolor{mainColor}
        \colorlet{symbolcolor}{white}
        \pgfmathtruncatemacro{\symindex}{mod(\i*7 + \j*5, 20)} % 20 símbolos
      \else
        \def\fillcolor{teste}
        \def\drawcolor{mainColor}
      \fi

      \begin{scope}[shift={(\x,\y)}]
        \filldraw[fill=\fillcolor, draw=\drawcolor, thick]
          (0:\R) \foreach \a in {60,120,...,300} { -- (\a:\R) } -- cycle;

        % Adiciona símbolo se for dourado
        \ifnum\isdourado=1
        \pgfmathtruncatemacro{\angle}{mod(\i*7+\j*11, 10)-5}
        % \pgfmathtruncatemacro{\angle}{mod(\i*11+\j*17, 20)-10}
        \edef\symbolcontent{\getsymbol{\symindex}}
        \node[rotate=\angle, text=mainColor] at (0,0) {\Large$\symbolcontent$};
        \fi
      \end{scope}
        }
    }

    \end{scope}
